from datetime import datetime import pandas as pd # 1. Dati di esempio delle spese spese_mese = [ { "Data": "2026-09-02", "Descrizione": "Spesa alimentare Supermercato", "Categoria": "Alimentari", "Importo (€)": 85.50, }, { "Data": "2026-09-05", "Descrizione": "Rifornimento Carburante", "Categoria": "Trasporti", "Importo (€)": 50.00, }, { "Data": "2026-09-10", "Descrizione": "Bolletta Elettricità", "Categoria": "Utenze", "Importo (€)": 120.30, }, { "Data": "2026-09-15", "Descrizione": "Cena Ristorante", "Categoria": "Svago", "Importo (€)": 45.00, }, { "Data": "2026-09-18", "Descrizione": "Acquisto Farmacia", "Categoria": "Salute", "Importo (€)": 24.80, }, ] df_spese = pd.DataFrame(spese_mese) df_spese["Data"] = pd.to_datetime(df_spese["Data"]) # 2. Assegnazione del numero della settimana df_spese["Settimana"] = df_spese["Data"].dt.isocalendar().week # 3. Visualizzazione Registro Dettaglio con la colonna Settimana print("=" * 70) print(" REGISTRO DETTAGLIO SPESE (CON SETTIMANA) ") print("=" * 70) print(df_spese.to_string(index=False)) # 4. Conteggi aggregati per Singola Settimana conteggi_settimanali = ( df_spese.groupby("Settimana")["Importo (€)"] .agg( N_Operazioni="count", Totale_Euro="sum", Media_Euro="mean", Max_Euro="max", ) .reset_index() ) print("\n" + "=" * 70) print(" CONTEGGI PER SINGOLA SETTIMANA ") print("=" * 70) print(conteggi_settimanali.to_string(index=False)) # 5. Dettaglio incrociato: Settimana x Categoria settimana_categoria = ( df_spese.groupby(["Settimana", "Categoria"])["Importo (€)"] .sum() .unstack(fill_value=0) ) print("\n" + "-" * 70) print(" SPESE DELLE SETTIMANE DIVISE PER CATEGORIA ") print("-" * 70) print(settimana_categoria) print("=" * 70)
