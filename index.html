import pandas as pd spese_mese = [ {"Data": "2026-09-02", "Descrizione": "Spesa alimentare Supermercato", "Categoria": "Alimentari", "Importo (€)": 85.50}, {"Data": "2026-09-05", "Descrizione": "Rifornimento Carburante", "Categoria": "Trasporti", "Importo (€)": 50.00}, {"Data": "2026-09-10", "Descrizione": "Bolletta Elettricità", "Categoria": "Utenze", "Importo (€)": 120.30}, {"Data": "2026-09-15", "Descrizione": "Cena Ristorante", "Categoria": "Svago", "Importo (€)": 45.00}, {"Data": "2026-09-18", "Descrizione": "Acquisto Farmacia", "Categoria": "Salute", "Importo (€)": 24.80} ] df_spese = pd.DataFrame(spese_mese) df_spese["Data_Dt"] = pd.to_datetime(df_spese["Data"]) df_spese["Settimana"] = "Settimana " + df_spese["Data_Dt"].dt.isocalendar().week.astype(str) totale_mese = df_spese["Importo (€)"].sum() spese_per_settimana = df_spese.groupby("Settimana")["Importo (€)"].sum().reset_index() spese_per_categoria = df_spese.groupby("Categoria")["Importo (€)"].sum().reset_index() print("============================================================") print(" REGISTRO DETTAGLIATO SPESE MESE ") print("============================================================") print(df_spese[["Data", "Settimana", "Descrizione", "Categoria", "Importo (€)"]].to_string(index=False)) print("\n============================================================") print(" RIEPILOGO SETTIMANALE ") print("============================================================") print(spese_per_settimana.to_string(index=False)) print("\n============================================================") print(" RIEPILOGO PER CATEGORIA ") print("============================================================") print(spese_per_categoria.to_string(index=False)) print("------------------------------------------------------------") print(f"TOTALE COMPLESSIVO DEL MESE: € {totale_mese:.2f}") print("============================================================")
