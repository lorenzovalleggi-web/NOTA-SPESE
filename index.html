from datetime import datetime import pandas as pd # 1. Definizione della struttura dati per le spese spese_mese = [ { "Data": "2026-09-02", "Descrizione": "Spesa alimentare Supermercato", "Categoria": "Alimentari", "Importo (€)": 85.50, }, { "Data": "2026-09-05", "Descrizione": "Rifornimento Carburante", "Categoria": "Trasporti", "Importo (€)": 50.00, }, { "Data": "2026-09-10", "Descrizione": "Bolletta Elettricità", "Categoria": "Utenze", "Importo (€)": 120.30, }, { "Data": "2026-09-15", "Descrizione": "Cena Ristorante", "Categoria": "Svago", "Importo (€)": 45.00, }, { "Data": "2026-09-18", "Descrizione": "Acquisto Farmacia", "Categoria": "Salute", "Importo (€)": 24.80, }, ] # 2. Creazione del DataFrame Pandas df_spese = pd.DataFrame(spese_mese) df_spese["Data"] = pd.to_datetime(df_spese["Data"]) # 3. Visualizzazione del REGISTRO DETTAGLIATO print("=" * 60) print(" REGISTRO DETTAGLIATO SPESE MESE ") print("=" * 60) print(df_spese.to_string(index=False)) # 4. Calcolo Totale Mensile e Riassunto per Categoria totale_mese = df_spese["Importo (€)"].sum() spese_per_categoria = ( df_spese.groupby("Categoria")["Importo (€)"].sum().reset_index() ) print("\n" + "=" * 60) print(" RIEPILOGO E VISIONE D'INSIEME ") print("=" * 60) print(spese_per_categoria.to_string(index=False)) print("-" * 60) print(f"TOTALE COMPLESSIVO DEL MESE: € {totale_mese:.2f}") print("=" * 60) # 5.
