from datetime import datetime import pandas as pd # 1. Dati di tutte le operazioni del mese spese_mese = [ { "Data": "2026-09-02", "Descrizione": "Spesa alimentare Supermercato", "Categoria": "Alimentari", "Importo (€)": 85.50, }, { "Data": "2026-09-05", "Descrizione": "Rifornimento Carburante", "Categoria": "Trasporti", "Importo (€)": 50.00, }, { "Data": "2026-09-10", "Descrizione": "Bolletta Elettricità", "Categoria": "Utenze", "Importo (€)": 120.30, }, { "Data": "2026-09-15", "Descrizione": "Cena Ristorante", "Categoria": "Svago", "Importo (€)": 45.00, }, { "Data": "2026-09-18", "Descrizione": "Acquisto Farmacia", "Categoria": "Salute", "Importo (€)": 24.80, }, ] df_spese = pd.DataFrame(spese_mese) df_spese["Data"] = pd.to_datetime(df_spese["Data"]) # Aggiunta colonna Settimana per chiarezza df_spese["Settimana"] = df_spese["Data"].dt.isocalendar().week # ============================================================================== # PAGINA 2: REGISTRO DETTAGLIATO (TUTTE LE OPERAZIONI DEL MESE) # ============================================================================== print("=" * 75) print("PAGINA 2 - REGISTRO DETTAGLIATO COMPLETO DEL MESE") print("=" * 75) # Impostazione per mostrare TUTTE le righe e colonne senza tagli pd.set_option("display.max_rows", None) pd.set_option("display.max_columns", None) pd.set_option("display.width", 1000) # Stampa del registro completo print(df_spese.to_string(index=False)) print("-" * 75) print(f"TOTALE OPERAZIONI REGISTRATE: {len(df_spese)}") print(f"TOTALE IMPORTO MESE: € {df_spese['Importo (€)'].sum():.2f}") print("=" * 75) # ============================================================================== # ESPORTAZIONE IN EXCEL SU DUE PAGINE/FOGLI SEPARATI # ============================================================================== # Conteggi per Pagina 1 conteggi_settimanali = ( df_spese.groupby("Settimana")["Importo (€)"] .agg(N_Operazioni="count", Totale_Euro="sum", Media_Euro="mean") .reset_index() ) with pd.ExcelWriter("Report_Spese_Mensile.xlsx", engine="openpyxl") as writer: # Pagina 1: Solo Riepiloghi e Conteggi conteggi_settimanali.to_excel( writer, sheet_name="Pagina 1 - Riepilogo", index=False ) # Pagina 2: Registro Dettagliato con TUTTE le operazioni del mese df_spese.to_excel( writer, sheet_name="Pagina 2 - Registro Dettaglio", index=False ) print( "\n-> Il file Excel 'Report_Spese_Mensile.xlsx' è stato creato con la Pagina 2 contenente l'elenco visivo completo di tutte le spese!" )
