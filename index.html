import pandas as pd # 1. Elenco delle spese del mese spese_mese = [ { "Data": "2026-09-02", "Descrizione": "Spesa alimentare Supermercato", "Categoria": "Alimentari", "Importo (€)": 85.50, }, { "Data": "2026-09-05", "Descrizione": "Rifornimento Carburante", "Categoria": "Trasporti", "Importo (€)": 50.00, }, { "Data": "2026-09-10", "Descrizione": "Bolletta Elettricità", "Categoria": "Utenze", "Importo (€)": 120.30, }, { "Data": "2026-09-15", "Descrizione": "Cena Ristorante", "Categoria": "Svago", "Importo (€)": 45.00, }, { "Data": "2026-09-18", "Descrizione": "Acquisto Farmacia", "Categoria": "Salute", "Importo (€)": 24.80, }, ] # 2. Creazione della tabella (DataFrame) df_spese = pd.DataFrame(spese_mese) df_spese["Data"] = pd.to_datetime(df_spese["Data"]).dt.strftime("%Y-%m-%d") # 3. Calcolo dei totali totale_mese = df_spese["Importo (€)"].sum() spese_per_categoria = ( df_spese.groupby("Categoria")["Importo (€)"].sum().reset_index() ) # 4. Stampa a schermo dei risultati print("=" * 60) print(" REGISTRO DETTAGLIATO SPESE MESE ") print("=" * 60) print(df_spese.to_string(index=False)) print("\n" + "=" * 60) print(" RIEPILOGO PER CATEGORIA ") print("=" * 60) print(spese_per_categoria.to_string(index=False)) print("-" * 60) print(f"TOTALE COMPLESSIVO DEL MESE: € {totale_mese:.2f}") print("=" * 60)
